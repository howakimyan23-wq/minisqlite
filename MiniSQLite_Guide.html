cd /home/shavarsh/minisqlite

Օրինակներ

Աղյուսակ և տվյալներ

CREATE TABLE students (id INTEGER PRIMARY KEY, name TEXT, city TEXT);
INSERT INTO students VALUES (26, 'Նոր Ուսանող', 'Երևան');
SELECT * FROM students;
SELECT * FROM students WHERE city = 'Երևան';

Փոփոխում և ջնջում

UPDATE students SET city = 'Գյումրի' WHERE id = 26;
DELETE FROM students WHERE id = 26;

Հարցումներ

SELECT DISTINCT city FROM students ORDER BY city ASC LIMIT 5;
SELECT city, COUNT(*) FROM students GROUP BY city;
SELECT SUM(id), AVG(id), MIN(id), MAX(id) FROM students;
